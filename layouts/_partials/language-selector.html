{{/* Project override of the theme's language-selector partial. Identical to the theme version except the menu variant renders no hover dropdown: with only two languages the trigger already links straight to the other one, so the popup is pure noise. base.js still toggles is-open on hover, but with no
    in the markup there is nothing to open. */ -}} {{- $page := .page -}} {{- $class := .class | default "" -}} {{- $mode := .mode | default "menu" -}} {{- $shellIcon := eq (.icon | default "") "shell" -}} {{- $targets := partial "language-targets.html" $page -}} {{- $count := len $targets -}} {{- if gt $count 1 -}} {{- $currentIndex := 0 -}} {{- range $index, $target := $targets -}} {{- if $target.active }}{{ $currentIndex = $index }}{{ end -}} {{- end -}} {{- $next := index $targets (mod (add $currentIndex 1) $count) -}} {{- if eq $mode "list" -}}
    {{- range $targets -}} {{- if .active -}} {{ .label }} {{- else -}} {{ .label }} {{- end -}} {{- end }}
    {{- else -}}
    {{- if $shellIcon -}} {{- partialCached "shell/icon.html" "languages" "languages" -}} {{- else }} {{- end }} {{ $next.label }}
    {{- end -}} {{- end -}}
